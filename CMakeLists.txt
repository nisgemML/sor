cmake_minimum_required(VERSION 3.22)
project(smart_order_router VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sor_iface INTERFACE)
target_include_directories(sor_iface INTERFACE include)
target_compile_options(sor_iface INTERFACE -Wall -Wextra)
enable_testing()
add_executable(test_router tests/test_router.cpp)
target_link_libraries(test_router PRIVATE sor_iface)
add_test(NAME Router COMMAND test_router)
